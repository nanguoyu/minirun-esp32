cmake_minimum_required(VERSION 3.20)
project(minirun_host LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(LLAMA_ROOT "${CMAKE_SOURCE_DIR}/deps/llama.cpp" CACHE PATH "llama.cpp source directory")
option(MINIRUN_SANITIZE "Run host tests with address and undefined-behavior sanitizers" OFF)
option(MINIRUN_MODEL_TESTS "Include tests that read artifacts/model.gguf" OFF)
if(NOT EXISTS "${LLAMA_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Run bash tools/setup.sh first to fetch llama.cpp")
endif()
foreach(feature COMMON TESTS TOOLS EXAMPLES SERVER APP UI)
    set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" "llama" EXCLUDE_FROM_ALL)

function(minirun_target target)
    target_include_directories(${target} PRIVATE src)
    target_compile_options(${target} PRIVATE -ffp-contract=off -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(${target} PRIVATE m)
    if(MINIRUN_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
foreach(target host host-aligned)
    add_executable(${target} src/host.c src/lfm.c)
    minirun_target(${target})
endforeach()
target_compile_definitions(host-aligned PRIVATE LFM_ALIGNED_MODEL=1)
add_executable(reference tools/reference.cpp)
target_link_libraries(reference PRIVATE llama)
if(APPLE)
    enable_language(OBJC)
    add_executable(render_text tools/render_text.m)
    target_link_libraries(render_text PRIVATE "-framework Cocoa")
endif()

enable_testing()
add_executable(test_core tests/test_core.c src/lfm.c)
minirun_target(test_core)
target_link_libraries(test_core PRIVATE ggml-base)
add_test(NAME core COMMAND test_core)
foreach(bits 16 32)
    foreach(kind file shared)
        set(target test_kv_${kind}_${bits})
        add_executable(${target} tests/test_kv_${kind}.c src/lfm.c src/kv_file.c)
        minirun_target(${target})
        target_compile_definitions(${target} PRIVATE LFM_CONTEXT=64 LFM_KV_BITS=${bits} LFM_KV_SHARED=1)
        add_test(NAME kv_${kind}_${bits} COMMAND ${target})
    endforeach()
endforeach()
if(MINIRUN_MODEL_TESTS)
    foreach(kind prefill context kv_model)
        add_executable(test_${kind} tests/test_${kind}.c src/lfm.c src/kv_file.c)
        minirun_target(test_${kind})
        target_compile_definitions(test_${kind} PRIVATE LFM_CONTEXT=128 LFM_KV_SHARED=1)
        add_test(NAME ${kind} COMMAND test_${kind} "${CMAKE_SOURCE_DIR}/artifacts/model.gguf" "${CMAKE_BINARY_DIR}/${kind}")
    endforeach()
    # test_prefill takes only the model path.
    set_tests_properties(prefill PROPERTIES DISABLED TRUE)
    add_test(NAME prefill_batches COMMAND test_prefill "${CMAKE_SOURCE_DIR}/artifacts/model.gguf")
endif()
